### **Промяна #3: JavaScript функционалност**
```javascript
// Get Level 1 assemblies (direct children of root)
function getLevel1Assemblies() {
    const level1 = [];
    
    if (bomData && bomData.assemblies) {
        bomData.assemblies.forEach(assembly => {
            // Level 1 assemblies have only one '/' in their path (after the root)
            const pathParts = assembly.path.split('/');
            if (pathParts.length === 2 && pathParts[0] && pathParts[1]) {
                level1.push({
                    name: assembly.name,
                    path: assembly.path,
                    children: assembly.children || []
                });
            }
        });
    }
    
    return level1;
}

// Classification wizard functionality
let currentWizardData = {
    assemblies: [],
    currentIndex: 0,
    classifications: {}
};

function startClassificationWizard() {
    const level1Assemblies = getLevel1Assemblies();
    
    if (level1Assemblies.length === 0) {
        alert('Няма намерени сборки от ниво 1 за класификация.');
        return;
    }
    
    currentWizardData = {
        assemblies: level1Assemblies,
        currentIndex: 0,
        classifications: loadExistingClassifications()
    };
    
    showWizardStep();
    document.getElementById('classificationWizard').classList.remove('hidden');
}

function loadExistingClassifications() {
    const classifications = { workshop: [], external: [] };
    
    for (let key in localStorage) {
        if (key.startsWith('classification_')) {
            const path = key.replace('classification_', '');
            const category = localStorage[key];
            if (category === 'workshop' || category === 'external') {
                classifications[category].push(path);
            }
        }
    }
    
    return classifications;
}

function showWizardStep() {
    const data = currentWizardData;
    const assembly = data.assemblies[data.currentIndex];
    
    if (!assembly) return;
    
    // Update UI
    document.getElementById('currentAssemblyName').textContent = assembly.name;
    document.getElementById('currentAssemblyPath').textContent = assembly.path;
    document.getElementById('wizardCurrentStep').textContent = data.currentIndex + 1;
    document.getElementById('wizardTotalSteps').textContent = data.assemblies.length;
    
    // Update navigation buttons
    document.getElementById('wizardPrev').disabled = data.currentIndex === 0;
    document.getElementById('wizardNext').disabled = true; // Will be enabled after classification
    document.getElementById('wizardFinish').classList.toggle('hidden', data.currentIndex < data.assemblies.length - 1);
    
    // Clear previous selections
    document.querySelectorAll('.classification-btn').forEach(btn => {
        btn.style.background = '';
        btn.style.color = '';
        btn.style.borderColor = '';
    });
    
    // Show existing classification if any
    const existingClassification = getExistingClassification(assembly.path);
    if (existingClassification) {
        highlightClassificationButton(existingClassification);
        document.getElementById('wizardNext').disabled = false;
    }
}

function getExistingClassification(assemblyPath) {
    if (currentWizardData.classifications.workshop.includes(assemblyPath)) {
        return 'workshop';
    }
    if (currentWizardData.classifications.external.includes(assemblyPath)) {
        return 'external';
    }
    return null;
}

function highlightClassificationButton(classification) {
    const btn = document.getElementById(classification === 'workshop' ? 'classifyWorkshop' : 'classifyExternal');
    if (btn) {
        btn.click(); // Simulate click to apply styles
    }
}

function classifyAssembly(category) {
    const assembly = currentWizardData.assemblies[currentWizardData.currentIndex];
    if (!assembly) return;
    
    // Remove from other category
    const otherCategory = category === 'workshop' ? 'external' : 'workshop';
    const otherIndex = currentWizardData.classifications[otherCategory].indexOf(assembly.path);
    if (otherIndex > -1) {
        currentWizardData.classifications[otherCategory].splice(otherIndex, 1);
    }
    
    // Add to selected category
    if (!currentWizardData.classifications[category].includes(assembly.path)) {
        currentWizardData.classifications[category].push(assembly.path);
    }
    
    // Save to localStorage
    localStorage.setItem('classification_' + assembly.path, category);
    
    // Enable next button
    document.getElementById('wizardNext').disabled = false;
    
    // Update button styles
    document.querySelectorAll('.classification-btn').forEach(btn => {
        btn.style.background = '';
        btn.style.color = '';
    });
    
    const selectedBtn = document.getElementById(category === 'workshop' ? 'classifyWorkshop' : 'classifyExternal');
    if (category === 'workshop') {
        selectedBtn.style.background = '#e67e22';
        selectedBtn.style.color = 'white';
    } else {
        selectedBtn.style.background = '#27ae60';
        selectedBtn.style.color = 'white';
    }
}

function nextWizardStep() {
    if (currentWizardData.currentIndex < currentWizardData.assemblies.length - 1) {
        currentWizardData.currentIndex++;
        showWizardStep();
    }
}

function prevWizardStep() {
    if (currentWizardData.currentIndex > 0) {
        currentWizardData.currentIndex--;
        showWizardStep();
    }
}

function skipWizardStep() {
    nextWizardStep();
}

function finishClassificationWizard() {
    // Close wizard
    document.getElementById('classificationWizard').classList.add('hidden');
    
    // Show success message
    const totalClassified = currentWizardData.classifications.workshop.length + 
                          currentWizardData.classifications.external.length;
    alert(`✅ Класификацията е запазена!\n\nОбщо класифицирани: ${totalClassified}\n🏭 В цеха: ${currentWizardData.classifications.workshop.length}\n🏗️ На обекта: ${currentWizardData.classifications.external.length}`);
    
    // Refresh the display if needed
    if (typeof updateClassificationDisplay === 'function') {
        updateClassificationDisplay();
    }
}

// Event listeners for wizard
document.addEventListener('DOMContentLoaded', function() {
    // Classification buttons
    document.getElementById('classifyWorkshop')?.addEventListener('click', () => classifyAssembly('workshop'));
    document.getElementById('classifyExternal')?.addEventListener('click', () => classifyAssembly('external'));
    
    // Navigation buttons
    document.getElementById('wizardPrev')?.addEventListener('click', prevWizardStep);
    document.getElementById('wizardNext')?.addEventListener('click', nextWizardStep);
    document.getElementById('wizardSkip')?.addEventListener('click', skipWizardStep);
    document.getElementById('wizardFinish')?.addEventListener('click', finishClassificationWizard);
    
    // Close wizard on outside click
    document.getElementById('classificationWizard')?.addEventListener('click', function(e) {
        if (e.target === this) {
            this.classList.add('hidden');
        }
    });
});
```

### 🎯 **Стъпка 2: Логика за detection**

**Функция `findParentAssembly(assemblyPath)`:**
```javascript
function findParentAssembly(assemblyPath) {
    const pathParts = assemblyPath.split('/');
    if (pathParts.length <= 1) return null;
    
    const parentPath = pathParts.slice(0, -1).join('/');
    const parentName = pathParts[pathParts.length - 2];
    
    // Determine category
    let category = null;
    if (assemblyClassification.workshop.includes(parentPath)) {
        category = 'workshop';
    } else if (assemblyClassification.external.includes(parentPath)) {
        category = 'external';
    }
    
    return { path: parentPath, name: parentName, category: category };
}
```

**Функция `findOrphanAssemblies(assemblyPath)`:**
```javascript
function findOrphanAssemblies(assemblyPath) {
    const parent = findParentAssembly(assemblyPath);
    if (!parent || !parent.category) return [];
    
    const orphans = [];
    const allPaths = [...assemblyClassification.workshop, ...assemblyClassification.external];
    
    allPaths.forEach(path => {
        if (path.startsWith(assemblyPath + '/')) {
            const pathParent = findParentAssembly(path);
            if (pathParent && pathParent.path === assemblyPath) {
                let childCategory = null;
                if (assemblyClassification.workshop.includes(path)) {
                    childCategory = 'workshop';
                } else if (assemblyClassification.external.includes(path)) {
                    childCategory = 'external';
                }
                
                if (childCategory && childCategory !== parent.category) {
                    orphans.push({
                        path: path,
                        name: extractNameFromPath(path),
                        childCategory: childCategory,
                        parentCategory: parent.category
                    });
                }
            }
        }
    });
    
    return orphans;
}
```

### 🎯 **Стъпка 4: Навигационна функция**

**`scrollToAssembly(assemblyPath, targetCategory)`:**
```javascript
function scrollToAssembly(assemblyPath, targetCategory = null) {
    let containerSelector = '';
    if (targetCategory === 'workshop') {
        containerSelector = '#workshopContent';
    } else if (targetCategory === 'external') {
        containerSelector = '#externalContent';
    }
    
    const pathParts = assemblyPath.split('/');
    const assemblyName = pathParts[pathParts.length - 1];
    
    let targetElement = null;
    const containers = containerSelector ? [document.querySelector(containerSelector)] : 
                      [document.querySelector('#workshopContent'), document.querySelector('#externalContent')];
    
    containers.forEach(container => {
        if (container && !targetElement) {
            const elements = container.querySelectorAll(`[data-path="${assemblyPath}"], [data-name="${assemblyName}"]`);
            if (elements.length > 0) {
                targetElement = elements[0];
            }
        }
    });
    
    if (targetElement) {
        expandAncestors(assemblyPath, containerSelector);
        setTimeout(() => {
            targetElement.scrollIntoView({ 
                behavior: 'smooth', 
                block: 'center' 
            });
            highlightAssembly(targetElement);
        }, 300);
    }
}
```

**`expandAncestors(assemblyPath, containerSelector)`:**
```javascript
function expandAncestors(assemblyPath, containerSelector = '') {
    const pathParts = assemblyPath.split('/');
    
    for (let i = 1; i < pathParts.length; i++) {
        const ancestorPath = pathParts.slice(0, i + 1).join('/');
        const containers = containerSelector ? [document.querySelector(containerSelector)] : 
                          [document.querySelector('#workshopContent'), document.querySelector('#externalContent')];
        
        containers.forEach(container => {
            if (container) {
                const ancestorElement = container.querySelector(`[data-path="${ancestorPath}"]`);
                if (ancestorElement) {
                    const childrenDiv = ancestorElement.parentElement.querySelector('.children');
                    const expandButton = ancestorElement.querySelector('.expand-button');
                    
                    if (childrenDiv && childrenDiv.style.display === 'none' && expandButton) {
                        expandButton.click();
                    }
                }
            }
        });
    }
}
```

**`highlightAssembly(element)`:**
```javascript
function highlightAssembly(element) {
    element.classList.add('assembly-highlight');
    setTimeout(() => {
        element.classList.remove('assembly-highlight');
    }, 3000);
}
```

## 🐛 Проблеми и решения

### Проблем 1: ⚠️ Иконата за root level сборки
**Симптом:** "Буфер кухня елементи" показва ⚠️ икона "ALL Bufer Service Komplekt"  
**Причина:** Root level сборките се класифицират, но не са показвани  
**Решение:** Добавена проверка `!parentPath.includes('/')`

### Проблем 2: parentName is not defined
**Симптом:** Runtime error на ред 2730  
**Причина:** Използване `parentName` преди дефиниция или е undefined  
**Решение:** Добавен код `const parentName = extractNameFromPath(parentPath);`

### Проблем 3: Излишни класификации в новия JSON
**Причина:** При смяна на JSON файл, старите localStorage записи остават  
**Следствие:** localStorage съдържа класификации, които не съответстват в новия BOM  
**Решение:** Имплементирана `validateAndCleanClassification()` функция

### Проблем 4: Scroll към грешна сборка при различна категория
**Причина:** [👁️ Виж] търсене "M12x30", а то е скрито  
**Следствие:** `querySelector` намира първия елемент, независимо  
**Решение:** Category-aware scrolling - търси в правилната секция
